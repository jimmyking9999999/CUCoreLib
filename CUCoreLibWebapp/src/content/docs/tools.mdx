```csharp panel title="UsingTheResults.cs"
// Colors from the color picker go anywhere a Unity Color is expected
LiquidRegistry.Register("glowjuice", new CustomLiquidInfo
{
    name = "Glow juice",
    color = new Color(0.5f, 0.88f, 0.63f)
});

// Layer masks go into SpawnLayers
BuildingEntityRegistry.Register("glowcap", new CustomBuildingEntityDefinition
{
    Name = "Glowcap cluster",
    Sprite = AssetLoader.LoadEmbeddedSprite("Images.glowcap.png"),
    SpawnMinPerChunk = 0.05f,
    SpawnMaxPerChunk = 0.08f,
    SpawnLayers = BuildingEntityRegistry.LayersToMask(3, 4)
});

// The loot weight is the 4th argument of ItemRegistry.Register
ItemRegistry.Register("glowcapspore", info, sporeSprite, 2);
```
